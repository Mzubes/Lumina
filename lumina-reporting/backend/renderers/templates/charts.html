{# Chart markup for print. Ports of the app's BarChart / CompositionBar / DonutChart, with the hover layer removed and everything it carried promoted onto the page: a legend for two or more series, a direct value label on every mark. That is not politeness -- three palette hues sit below 3:1 against paper, and visible labels are what makes them legal. Every colour reaching these macros is already a literal hex resolved in Python. Nothing here may use color-mix() or aspect-ratio; the print-CSS lint fails the build if it does. #} {% macro legend(entries) %} {% if entries %}
{% for entry in entries %} {{ entry.label }}{% if entry.value_label %} · {{ entry.value_label }}{% endif %} {% endfor %}
{% endif %} {% endmacro %} {% macro bar_chart(chart) %}
{% for row in chart.rows %}
{{ row.label }}
{% for bar in row.bars %} {% if chart.signed %} {# Diverging: zero down the middle, bars growing away from it. A negative drawn as a positive-length bar is a false chart. #}
{% if bar.negative %} {{ bar.value_label }} {% endif %} {% if not bar.negative %} {{ bar.value_label }} {% endif %}
{% else %}
{# The direct label the screen component puts in a tooltip. #} {{ bar.value_label }}
{% endif %} {% endfor %}
{% endfor %} {{ legend(chart.legend) }}
{% endmacro %} {% macro composition_chart(chart) %}
{% for segment in chart.segments %} {# flex-grow, so the segments always sum to exactly the track width however the percentages round. #} {% endfor %}
{{ legend(chart.legend) }}
{% endmacro %} {% macro donut_chart(chart) %}
{{ legend(chart.legend) }}
{% endmacro %} {% macro chart(model) %} {% if model.kind == 'bar' %}{{ bar_chart(model) }} {% elif model.kind == 'composition' %}{{ composition_chart(model) }} {% elif model.kind == 'donut' %}{{ donut_chart(model) }} {% endif %} {% endmacro %}
